def extract(source,crop,flip=False,color_mode=None):
 im=Image.open(SOURCES/source).convert('RGBA').crop(tuple(crop));a=np.array(im)
 # Select the declared inspected component. Dilate the shape support 3px to retain soft edge alpha.
 if color_mode=='sage-worn':
  rgb=a[:,:,:3].astype(int); core=(rgb[:,:,1]-rgb[:,:,0]>5)&(rgb[:,:,1]-rgb[:,:,2]>8)&(a[:,:,3]>96)
  dark=(rgb.max(axis=2)<150)
  support=binary_fill_holes(core|(binary_dilation(core,iterations=6)&dark))
  # One-pixel soft mask keeps extracted internal boundaries antialiased without keeping body-colored fringe.
  soft=Image.fromarray((support*255).astype('uint8')).filter(ImageFilter.GaussianBlur(0.35))
  a[:,:,3]=(a[:,:,3].astype(float)*np.array(soft)/255).round().astype('uint8')
  support=binary_dilation(support,iterations=1)
 else:
  lab,n=label(a[:,:,3]>96);counts=np.bincount(lab.ravel());counts[0]=0
  if not counts.any(): raise ValueError((source,crop))
  support=binary_dilation(lab==counts.argmax(),iterations=3)
 a[:,:,3]=np.where(support,a[:,:,3],0)
 a[a[:,:,3]==0,:3]=0
 im=Image.fromarray(a)
 box=im.getbbox();im=im.crop(box)
 if flip:im=im.transpose(Image.Transpose.FLIP_LEFT_RIGHT)
 return im,[crop[0]+box[0],crop[1]+box[1],crop[0]+box[2],crop[1]+box[3]]
def fit(im,width,center,top=None,bottom=None):
 h=round(im.height*width/im.width);im=im.resize((width,h),Image.Resampling.LANCZOS)
 # Discard Lanczos nonphysical alpha ringing only below 2/255, keep genuine soft alpha.
 a=np.array(im);a[a[:,:,3]<2]=0;im=Image.fromarray(a)
 x=round(center-width/2); y=top if top is not None else bottom-h
 canvas=Image.new('RGBA',(512,512));canvas.alpha_composite(im,(x,y));return canvas,{'scale':width/im.width,'size':[width,h],'position':[x,y]}
def export_layer(item,view,name,im,bone,plane):
 bbox=im.getbbox()
 if not bbox:return None
 piece=im.crop(bbox);rel=f'assets/{item}/{view}-{name}.png';p=OUT/rel;p.parent.mkdir(parents=True,exist_ok=True);piece.save(p)
 rect=[round((bbox[0]-117.4)/4.2,6),round((bbox[1]-175)/4.2,6),round((bbox[2]-bbox[0])/4.2,6),round((bbox[3]-bbox[1])/4.2,6)]
 d={'src':rel,'bone':bone,'layer':plane,'rect':rect,'sourceRect':list(bbox),'sha256':sha(p)}
 return d

   placed,f=fit(raw,w,part['centers'][j],top=part.get('tops',[None]*4)[j],bottom=part.get('bottoms',[None]*4)[j]);f['uniformScale']=s
